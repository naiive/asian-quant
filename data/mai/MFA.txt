
VAR1:=REF((LOW+OPEN+CLOSE+HIGH)/4,1);
DIFF:=LOW-VAR1;
NUM:=SMA(ABS(DIFF),13,1);
DEN:=SMA(MAX(DIFF,0),10,1);
VAR2:=IF(DEN=0,0,NUM/DEN);
VAR3:=EMA(VAR2,10);
VAR4:=LLV(LOW,33);
VAR5:=EMA(IF(LOW<=VAR4,VAR3,0),3);

THRESHOLD:=0.01;
MFA:=ROUND(VAR5*100)/100;

ISRED:=MFA>REF(MFA,1) AND MFA>=THRESHOLD;
ISGREEN:=MFA<REF(MFA,1) AND REF(MFA,1)>=THRESHOLD;


洗:IF(ISRED,MFA,DRAWNULL),NODRAW,COLORF03A55;
进:IF(ISGREEN,MFA,DRAWNULL),NODRAW,COLOR00A870;
STICKLINE(ISRED,0,MFA,0.7,0),COLORF03A55;
