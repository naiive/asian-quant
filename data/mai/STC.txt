
MACDV:=EMA(CLOSE,FAST)-EMA(CLOSE,SLOW);

LL1:=LLV(MACDV,N);
RNG1:=HHV(MACDV,N)-LL1;
ST1:=IF(RNG1>0,(MACDV-LL1)/RNG1*100,50);
SM1:=SMA(ST1,100,AAA*100);

LL2:=LLV(SM1,N);
RNG2:=HHV(SM1,N)-LL2;
ST2:=IF(RNG2>0,(SM1-LL2)/RNG2*100,50);
STC:=SMA(ST2,100,AAA*100);

UP:=STC>REF(STC,1);
UPP:=REF(UP,1);

STCUP:IF(UP OR UPP,STC,DRAWNULL),COLORGREEN,LINETHICK2;
STCDN:IF(NOT(UP) OR NOT(UPP),STC,DRAWNULL),COLORRED,LINETHICK2;

L25:25,COLORGRAY,DOTLINE;
L75:75,COLORGRAY,DOTLINE;
